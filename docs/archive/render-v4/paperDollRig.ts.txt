/** Pixel coordinates in the original 1774 × 887 art space. Never viewport pixels. */
export type Point = readonly [number, number];
export type FitRow = {
  sourceY: number;
  targetY: number;
  left: number;
  right: number;
};
export type FitStrip = {
  sourceLeft: number;
  sourceRight: number;
  rows: FitRow[];
};
export type Triangle = readonly [Point, Point, Point];

/** An affine map for a single triangle of the garment's registered mesh. */
export function triangleMatrix(source: Triangle, target: Triangle) {
  const [[x0, y0], [x1, y1], [x2, y2]] = source;
  const [[u0, v0], [u1, v1], [u2, v2]] = target;
  const determinant = (x1 - x0) * (y2 - y0) - (x2 - x0) * (y1 - y0);
  if (Math.abs(determinant) < 0.00001)
    throw new Error("Degenerate garment triangle");
  const a = ((u1 - u0) * (y2 - y0) - (u2 - u0) * (y1 - y0)) / determinant;
  const c = ((x1 - x0) * (u2 - u0) - (x2 - x0) * (u1 - u0)) / determinant;
  const b = ((v1 - v0) * (y2 - y0) - (v2 - v0) * (y1 - y0)) / determinant;
  const d = ((x1 - x0) * (v2 - v0) - (x2 - x0) * (v1 - v0)) / determinant;
  return `matrix(${a} ${b} ${c} ${d} ${u0 - a * x0 - c * y0} ${v0 - b * x0 - d * y0})`;
}

export function stripTriangles(strip: FitStrip) {
  return strip.rows.slice(1).flatMap((bottom, index) => {
    const top = strip.rows[index];
    const source: readonly Point[] = [
      [strip.sourceLeft, top.sourceY],
      [strip.sourceRight, top.sourceY],
      [strip.sourceRight, bottom.sourceY],
      [strip.sourceLeft, bottom.sourceY],
    ];
    const target: readonly Point[] = [
      [top.left, top.targetY],
      [top.right, top.targetY],
      [bottom.right, bottom.targetY],
      [bottom.left, bottom.targetY],
    ];
    return [
      [0, 1, 2],
      [0, 2, 3],
    ].map((vertices) => ({
      source: vertices.map((v) => source[v]) as unknown as Triangle,
      target: vertices.map((v) => target[v]) as unknown as Triangle,
    }));
  });
}
