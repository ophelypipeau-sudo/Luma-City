      )}
    </g>
  );
}

function PaintedBase({
  characterId,
  category,
  material: m,
  fill,
  shade,
  metal,
}: {
  characterId: string;
  category: Exclude<ClothingCategory, "hair">;
  material: Material;
  fill: string;
  shade: string;
  metal: string;
}) {
  const female = characterId === "iris" || characterId === "june";
  const outline = {
    stroke: m.edge,
    strokeWidth: 0.85,
    strokeLinejoin: "round" as const,
  };
  if (category === "tops" && characterId === "iris")
    return (
      <g fill={fill} {...outline}>
        <path d="M205 190Q224 207 233 242L227 247Q217 214 201 196Z M298 188Q315 201 321 238L315 242Q309 211 294 193Z" />
        <path d={corset} />
      </g>
    );
  if (category === "tops" && characterId === "june")
    return <path d={cardigan} fill={fill} {...outline} />;
  if (category === "bottoms")
    return (
      <g
        transform={
          characterId === "june" ? "translate(0 76) scale(1 .84)" : undefined
        }
      >
        <path d={female ? skirt : trousers} fill={fill} {...outline} />
        {female ? (
          <>
            <path
              d="M216 351Q275 362 338 350L343 379Q278 391 207 379Z"
              fill={fill}
              {...outline}
            />
            <circle cx="308" cy="368" r="3.7" fill={metal} />
          </>
        ) : (
          <path
            d="M600 380Q670 395 738 381L741 398Q668 410 595 397Z"
            fill={shade}
            {...outline}
          />
        )}
      </g>
    );
  if (category === "shoes" && female)
    return (
      <>
        <g fill={fill} {...outline}>
          <path d={leftBoot} />
          <path d={rightBoot} />
          <path d={bootCuffs} />
        </g>
        <path
          d="M138 849Q162 872 226 861L228 872Q196 890 163 882L140 871Z M229 851Q257 875 336 864L334 878Q291 890 251 880L230 868Z"
          fill={m.shade}
          {...outline}
        />
        <path
          d="M175 617Q207 628 240 617M245 617Q274 628 300 617"
          fill="none"
          stroke={m.shade}
          strokeWidth="1.2"
          opacity=".5"
        />
      </>
    );
  if (category === "shoes")
    return (
      <>
        <path d={sneakers} fill={fill} {...outline} />
        <path
          d="M520 852Q554 873 625 856L624 871Q579 885 535 874L521 866Z M648 851Q689 874 767 859L766 873L720 882L665 873L648 862Z"
          fill="#ddd7cc"
          {...outline}
        />
      </>
    );
  return (
    <>
      <path d={jacket} fill={fill} {...outline} />
      <path
        d="M636 177Q655 191 681 171L686 205L679 399Q655 402 636 393L626 215Z"
        fill="#eee9df"
        {...outline}
      />
      {characterId === "noa" && (
        <path
          d="M579 166Q548 182 537 218L520 264Q503 299 502 326Q511 356 548 380L583 407L609 386L559 339Q551 328 560 309L589 235L608 185Z M722 171Q757 185 768 222L783 338L791 434L757 441L745 342L725 244L707 197Z"
          fill="#e5dccb"
          {...outline}
        />
      )}
    </>
  );
}

function BootLaces({
  centers,
  color,
  edge,
}: {
  centers: [number, number][];
  color: string;
  edge: string;
}) {
  return (
    <g fill="none" strokeLinecap="round">
      {centers.map(([x, y], i) => {
        const half = i < 4 ? 12 : i < 6 ? 9 : 10;
        const next = centers[i + 1];
        return (
          <g key={i}>
            <circle cx={x - half} cy={y} r="2.3" stroke={edge} fill={color} />
            <circle cx={x + half} cy={y} r="2.3" stroke={edge} fill={color} />
            {next && (
              <>
                <path
                  d={`M${x - half} ${y}L${next[0] + half} ${next[1]}M${x + half} ${y}L${next[0] - half} ${next[1]}`}
                  stroke={edge}
                  strokeWidth="2.5"
                />
                <path
                  d={`M${x - half} ${y}L${next[0] + half} ${next[1]}M${x + half} ${y}L${next[0] - half} ${next[1]}`}
                  stroke={color}
                  strokeWidth="1.5"
                />
              </>
            )}
          </g>
        );
      })}
    </g>
  );
}
